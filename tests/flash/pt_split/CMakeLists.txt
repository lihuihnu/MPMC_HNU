cmake_minimum_required(VERSION 3.21)
project(mpmc_pt_split_tests LANGUAGES CXX)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../modules/flash" flash-library)
enable_testing()
add_executable(mpmc_pt_split_tests split_test.cpp headers.cpp)
target_link_libraries(mpmc_pt_split_tests PRIVATE mpmc::flash)
set_target_properties(mpmc_pt_split_tests PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(mpmc_pt_split_tests PRIVATE /W4 /WX /permissive- /utf-8)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    target_compile_options(mpmc_pt_split_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
option(MPMC_SPLIT_ENABLE_SANITIZERS "Check PT split with ASan/UBSan" OFF)
if(MPMC_SPLIT_ENABLE_SANITIZERS)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang)$")
        message(FATAL_ERROR "PT split sanitizers require Linux GCC/Clang")
    endif()
    target_compile_options(mpmc_pt_split_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(mpmc_pt_split_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
set(cases rr_binary rr_zero_trace rr_extreme rr_boundary rr_domains_budget
    imposed_equivalent imposed_common_tangent imposed_domains_budget balanced_seed
    iterate_exact iterate_refine iterate_limits iterate_callbacks split_ideal split_zero_trace
    split_single split_initial_unknown split_final_unknown split_third_phase split_disappearance
    split_domains pr_nitrogen pr_carbon pr_order pr_ownership pr_failure headers
    gibbs_slope gibbs_roundoff pr_gibbs_progress result_guard)
foreach(case IN LISTS cases)
    add_test(NAME flash.pt_split.${case} COMMAND mpmc_pt_split_tests ${case})
    set_tests_properties(flash.pt_split.${case} PROPERTIES LABELS "flash;flash.pt_split;unit" TIMEOUT 60)
endforeach()
