cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(mpmc_pr76_model_registry LANGUAGES CXX)
endif()
if(NOT TARGET mpmc::pr76_solver_configuration)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../pr76_solver" solver-library)
endif()
find_package(Threads REQUIRED)
add_library(mpmc_pr76_model_registry STATIC ../src/pr76_model_registry.cpp ../src/model_handle_entropy.cpp)
add_library(mpmc::pr76_model_registry ALIAS mpmc_pr76_model_registry)
target_compile_features(mpmc_pr76_model_registry PUBLIC cxx_std_20)
target_link_libraries(mpmc_pr76_model_registry PUBLIC mpmc::pr76_solver_configuration Threads::Threads)
if(WIN32)
    target_link_libraries(mpmc_pr76_model_registry PRIVATE bcrypt)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    find_library(MPMC_MODEL_SECURITY_FRAMEWORK Security REQUIRED)
    target_link_libraries(mpmc_pr76_model_registry PRIVATE "${MPMC_MODEL_SECURITY_FRAMEWORK}")
endif()
set_target_properties(mpmc_pr76_model_registry PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(mpmc_pr76_model_registry PRIVATE /W4 /WX /permissive- /utf-8)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    target_compile_options(mpmc_pr76_model_registry PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
