cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(mpmc_mesh LANGUAGES CXX)
endif()

add_library(mpmc_mesh INTERFACE)
add_library(mpmc::mesh ALIAS mpmc_mesh)
target_compile_features(mpmc_mesh INTERFACE cxx_std_20)
target_include_directories(mpmc_mesh INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")

option(MPMC_MESH_WITH_HDF5 "Build the optional MRST face-mesh HDF5 adapter" OFF)
if(MPMC_MESH_WITH_HDF5)
    enable_language(C)
    find_package(HDF5 1.14.6 REQUIRED COMPONENTS C)
    add_library(mpmc_mesh_hdf5 STATIC src/face_mesh_hdf5.cpp)
    add_library(mpmc::mesh_hdf5 ALIAS mpmc_mesh_hdf5)
    target_link_libraries(mpmc_mesh_hdf5 PUBLIC mpmc::mesh PRIVATE HDF5::HDF5)
    if(WIN32 AND HDF5_USE_STATIC_LIBRARIES)
        target_link_libraries(mpmc_mesh_hdf5 PRIVATE shlwapi)
    endif()
    add_executable(mpmc_mesh_convert tools/mesh_convert.cpp)
    target_link_libraries(mpmc_mesh_convert PRIVATE mpmc::mesh_hdf5)
    foreach(target IN ITEMS mpmc_mesh_hdf5 mpmc_mesh_convert)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        endif()
    endforeach()
endif()
