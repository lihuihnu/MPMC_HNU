cmake_minimum_required(VERSION 3.21)
project(MPMC_HNU LANGUAGES CXX)

add_subdirectory(modules/ad)

option(MPMC_ENABLE_THERMODYNAMICS "Expose thermodynamics library targets from the project root" OFF)
option(MPMC_ENABLE_FLASH "Expose flash and its thermodynamics dependency from the project root" OFF)

set(mpmc_enabled_modules ad)
if(MPMC_ENABLE_THERMODYNAMICS OR MPMC_ENABLE_FLASH)
    add_subdirectory(modules/thermodynamics)
    list(APPEND mpmc_enabled_modules thermodynamics)
endif()
if(MPMC_ENABLE_FLASH)
    add_subdirectory(modules/flash)
    list(APPEND mpmc_enabled_modules flash)
endif()
message(STATUS "MPMC root library modules: ${mpmc_enabled_modules}")

include(CTest)
if(BUILD_TESTING)
    message(STATUS "MPMC root tests: AD arithmetic only; module suites use their standalone entries")
    add_subdirectory(tests/ad)
else()
    message(STATUS "MPMC root tests: disabled")
endif()
